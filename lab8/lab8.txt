EXERCÍCIO 1
------------------------------------------------------------------------------
Enunciado:
Escreva um programa que use myallocate para alocar e liberar memória
dinamicamente.

Código fonte (ex1_myallocate.c):

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <signal.h>
#include <sys/types.h>
#include <sys/wait.h>

static char bufferSaida[8192];
static void *memorypool;

void *myallocate(int n);
void mydeallocate(void *bloco);
void initmemorypool(void);
void resetmemorypool(void);
void testeOrdemInversa(char *a, char *c, int tamC);

int main (void)
{
    setvbuf(stdout, bufferSaida, _IOFBF, sizeof bufferSaida);

    initmemorypool();
    printf("pool iniciado: break inicial = %p\n\n", memorypool);

    int *a = (int *) myallocate(10 * sizeof(int));
    printf("a = myallocate(%zu)   -> %p | break = %p\n", 10 * sizeof(int), (void *) a, sbrk(0));

    char *b = (char *) myallocate(64);
    printf("b = myallocate(64)   -> %p | break = %p\n", (void *) b, sbrk(0));

    char *c = (char *) myallocate(8192);
    printf("c = myallocate(8192) -> %p | break = %p\n\n", (void *) c, sbrk(0));

    for (int i = 0; i < 10; i++)
        a[i] = i * i;
    strcpy(b, "memoria obtida direto do heap com sbrk()");
    memset(c, 'Z', 8192);

    printf("a[0..9] =");
    for (int i = 0; i < 10; i++)
        printf(" %d", a[i]);
    printf("\nb = \"%s\"\n", b);
    printf("c[0] = '%c', c[8191] = '%c'\n", c[0], c[8191]);
    printf("b - a = %ld bytes, c - b = %ld bytes (blocos contiguos)\n\n", (long) (b - (char *) a), (long) (c - b));
    fflush(stdout);

    testeOrdemInversa((char *) a, c, 8192);

    mydeallocate(c);
    printf("mydeallocate(c)   -> break = %p (igual a c: %s)\n", sbrk(0), sbrk(0) == (void *) c ? "sim" : "nao");
    mydeallocate(b);
    printf("mydeallocate(b)   -> break = %p (igual a b: %s)\n", sbrk(0), sbrk(0) == (void *) b ? "sim" : "nao");
    resetmemorypool();
    printf("resetmemorypool() -> break = %p (igual ao inicio do pool: %s)\n", sbrk(0), sbrk(0) == memorypool ? "sim" : "nao");

    return 0;
}

void *myallocate(int n)
{
    void *bloco = sbrk(n);
    return bloco == (void *) -1 ? NULL : bloco;
}

void mydeallocate(void *bloco)
{
    brk(bloco);
}

void initmemorypool(void)
{
    memorypool = sbrk(0);
}

void resetmemorypool(void)
{
    brk(memorypool);
}

void testeOrdemInversa(char *a, char *c, int tamC)
{
    pid_t pid;
    int status;

    pid = fork();
    if (pid < 0) {
        perror("fork");
        exit(1);
    }

    if (pid == 0) {
        printf("filho: liberando 'a' antes de 'b' e 'c' com brk(a)\n");
        fflush(stdout);
        mydeallocate(a);
        printf("filho: break agora em %p, lendo c[%d]...\n", sbrk(0), tamC - 1);
        fflush(stdout);
        volatile char x = c[tamC - 1];
        printf("filho: leitura funcionou (%c)\n", x);
        fflush(stdout);
        _exit(0);
    }

    waitpid(pid, &status, 0);
    if (WIFSIGNALED(status))
        printf("pai: filho terminou pelo sinal %d (%s)\n\n", WTERMSIG(status), WTERMSIG(status) == SIGSEGV ? "SIGSEGV" : "outro");
    else
        printf("pai: filho terminou normalmente com codigo %d\n\n", WEXITSTATUS(status));
    fflush(stdout);
}

Reflexão:
myallocate(n) chama sbrk(n): o break do heap avança n bytes e o valor
devolvido é o endereço antigo do break, ou seja, o início do bloco novo.
Por isso os blocos a, b e c saem contíguos (b - a = 40, c - b = 64). A
liberação com brk(bloco) devolve o break para o início do bloco, então só é
possível liberar em ordem inversa (LIFO), como numa pilha: liberar c, depois
b, depois o pool inteiro funcionou, e o break coincidiu com o endereço de
cada bloco. Quando o filho liberou "a" antes de "c", o break voltou para o
início de "a" e a leitura de c[8191] caiu em página já devolvida, resultando
em SIGSEGV (confirmado pelo strace: si_addr = c + 8191). O programa usa
setvbuf com buffer estático para a saída porque misturar printf (que usa
malloc) com sbrk manual poderia corromper o heap.

Enunciado:
Execute o programa map-exemplo.c passando um nome de arquivo como parâmetro.
Qual foi a novidade?

Código fonte (map_exemplo.c):

#include <stdio.h>
#include <stdlib.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>

int main (int argc, char *argv[])
{
    struct stat sb;
    off_t len;
    char *p;
    int fd;

    if (argc < 2) {
        fprintf(stderr, "usage: %s <file>\n", argv[0]);
        exit(1);
    }

    fd = open(argv[1], O_RDONLY);
    if (fd == -1) {
        perror("open");
        exit(1);
    }

    if (fstat(fd, &sb) == -1) {
        perror("fstat");
        exit(1);
    }

    if (!S_ISREG(sb.st_mode)) {
        fprintf(stderr, "%s is not a file\n", argv[1]);
        exit(1);
    }

    p = mmap(0, sb.st_size, PROT_READ, MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror("mmap");
        exit(1);
    }

    if (close(fd) == -1) {
        perror("close");
        exit(1);
    }

    for (len = 0; len < sb.st_size; len++)
        putchar(p[len]);

    if (munmap(p, sb.st_size) == -1) {
        perror("munmap");
        exit(1);
    }

    return 0;
}

EXERCÍCIO 3
------------------------------------------------------------------------------
Enunciado:
Modifique o programa map-exemplo.c para: (a) verificar quantas páginas de
memória foram necessárias para o arquivo; (b) medir quanto mais rápido é
escrever no arquivo mapeado do que fazer E/S em disco (dica: gettimeofday).

Código fonte (ex3_map_paginas_tempo.c):

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <sys/time.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>

#define REPETICOES 5
#define SAIDA_WRITE "saida_write.bin"
#define SAIDA_MMAP "saida_mmap.bin"

typedef struct {
    double semSync;
    double comSync;
} Tempos;

typedef struct {
    const char *nome;
    int usaMmap;
    size_t bloco;
} Cenario;

static long agoraUs(void);
static long paginasNoKernel(void *p, long pagesize);
static Tempos escreveWrite(const char *src, off_t n, size_t bloco);
static Tempos escreveMmap(const char *src, off_t n, size_t bloco);
static void mede(const char *src, off_t n, long pagesize);

int main (int argc, char *argv[])
{
    struct stat sb;
    long pagesize, paginas, noKernel;
    char *p;
    int fd;

    if (argc < 2) {
        fprintf(stderr, "usage: %s <file> [medir]\n", argv[0]);
        exit(1);
    }

    fd = open(argv[1], O_RDONLY);
    if (fd == -1) {
        perror("open");
        exit(1);
    }

    if (fstat(fd, &sb) == -1) {
        perror("fstat");
        exit(1);
    }

    if (!S_ISREG(sb.st_mode)) {
        fprintf(stderr, "%s is not a file\n", argv[1]);
        exit(1);
    }

    p = mmap(0, sb.st_size, PROT_READ, MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror("mmap");
        exit(1);
    }

    if (close(fd) == -1) {
        perror("close");
        exit(1);
    }

    pagesize = sysconf(_SC_PAGESIZE);
    paginas = (sb.st_size + pagesize - 1) / pagesize;
    noKernel = paginasNoKernel(p, pagesize);

    printf("arquivo: %s\n", argv[1]);
    printf("tamanho do arquivo: %ld bytes\n", (long) sb.st_size);
    printf("tamanho da pagina: %ld bytes\n", pagesize);
    printf("paginas necessarias: %ld\n", paginas);
    printf("bytes mapeados (paginas x tamanho): %ld\n", paginas * pagesize);
    printf("bytes sobrando na ultima pagina (preenchidos com zero): %ld\n", paginas * pagesize - (long) sb.st_size);
    printf("paginas do mapeamento segundo /proc/self/maps: %ld\n", noKernel);

    if (argc >= 3 && strcmp(argv[2], "medir") == 0)
        mede(p, sb.st_size, pagesize);

    if (munmap(p, sb.st_size) == -1) {
        perror("munmap");
        exit(1);
    }

    return 0;
}

static long agoraUs(void)
{
    struct timeval tv;

    gettimeofday(&tv, NULL);
    return tv.tv_sec * 1000000L + tv.tv_usec;
}

static long paginasNoKernel(void *p, long pagesize)
{
    FILE *f = fopen("/proc/self/maps", "r");
    char linha[512];
    unsigned long ini, fim;

    if (f == NULL)
        return -1;

    while (fgets(linha, sizeof linha, f) != NULL) {
        if (sscanf(linha, "%lx-%lx", &ini, &fim) == 2 && ini == (unsigned long) p) {
            fclose(f);
            return (long) ((fim - ini) / pagesize);
        }
    }

    fclose(f);
    return -1;
}

static Tempos escreveWrite(const char *src, off_t n, size_t bloco)
{
    Tempos t;
    long ini, meio, fim;
    off_t pos;
    size_t q;
    int fd;

    ini = agoraUs();

    fd = open(SAIDA_WRITE, O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd == -1) {
        perror("open saida");
        exit(1);
    }

    for (pos = 0; pos < n; pos += bloco) {
        q = (n - pos < (off_t) bloco) ? (size_t) (n - pos) : bloco;
        if (write(fd, src + pos, q) != (ssize_t) q) {
            perror("write");
            exit(1);
        }
    }

    meio = agoraUs();

    if (fsync(fd) == -1) {
        perror("fsync");
        exit(1);
    }

    fim = agoraUs();
    close(fd);

    t.semSync = (double) (meio - ini);
    t.comSync = (double) (fim - ini);
    return t;
}

static Tempos escreveMmap(const char *src, off_t n, size_t bloco)
{
    Tempos t;
    long ini, meio, fim;
    off_t pos;
    size_t q;
    char *d;
    volatile char *v;
    int fd;

    ini = agoraUs();

    fd = open(SAIDA_MMAP, O_RDWR | O_CREAT | O_TRUNC, 0644);
    if (fd == -1) {
        perror("open saida");
        exit(1);
    }

    if (ftruncate(fd, n) == -1) {
        perror("ftruncate");
        exit(1);
    }

    d = mmap(0, n, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (d == MAP_FAILED) {
        perror("mmap saida");
        exit(1);
    }

    if (bloco == 1) {
        v = d;
        for (pos = 0; pos < n; pos++)
            v[pos] = src[pos];
    } else {
        for (pos = 0; pos < n; pos += bloco) {
            q = (n - pos < (off_t) bloco) ? (size_t) (n - pos) : bloco;
            memcpy(d + pos, src + pos, q);
        }
    }

    meio = agoraUs();

    if (msync(d, n, MS_SYNC) == -1) {
        perror("msync");
        exit(1);
    }

    fim = agoraUs();
    munmap(d, n);
    close(fd);

    t.semSync = (double) (meio - ini);
    t.comSync = (double) (fim - ini);
    return t;
}

static void mede(const char *src, off_t n, long pagesize)
{
    Cenario cenarios[4] = {
        {"write(), 1 byte por chamada", 0, 1},
        {"write(), 1 pagina por chamada", 0, 0},
        {"mmap, escrita byte a byte", 1, 1},
        {"mmap, memcpy de 1 pagina por vez", 1, 0}
    };
    double semSync[4], comSync[4];
    volatile unsigned long soma = 0;
    off_t i;
    int c, r;

    for (i = 0; i < n; i++)
        soma += (unsigned char) src[i];

    printf("\nescrita de %ld bytes, media de %d repeticoes (tempos em microssegundos)\n", (long) n, REPETICOES);
    printf("%-36s %16s %16s\n", "cenario", "sem sync", "com fsync/msync");

    for (c = 0; c < 4; c++) {
        size_t bloco = cenarios[c].bloco == 0 ? (size_t) pagesize : cenarios[c].bloco;
        double somaSem = 0, somaCom = 0;

        for (r = 0; r < REPETICOES; r++) {
            Tempos t = cenarios[c].usaMmap ? escreveMmap(src, n, bloco) : escreveWrite(src, n, bloco);
            somaSem += t.semSync;
            somaCom += t.comSync;
        }

        semSync[c] = somaSem / REPETICOES;
        comSync[c] = somaCom / REPETICOES;
        printf("%-36s %16.0f %16.0f\n", cenarios[c].nome, semSync[c], comSync[c]);
    }

    printf("\nmmap byte a byte comparado com write() de 1 byte:  %.1f vezes mais rapido (sem sync), %.1f vezes (com sync)\n",
           semSync[0] / semSync[2], comSync[0] / comSync[2]);
    printf("mmap memcpy comparado com write() de 1 pagina:     %.1f vezes mais rapido (sem sync), %.1f vezes (com sync)\n",
           semSync[1] / semSync[3], comSync[1] / comSync[3]);
}

1 página (3870 bytes sobram, preenchidos com zero), 4096 bytes usa 1 página
sem sobra, 4097 bytes usa 2 páginas (4095 sobram) e 10000 bytes usa 3. O
valor calculado bateu com o número de páginas do mapeamento em
/proc/self/maps em todos os casos.
(b) Escrever byte a byte pelo mmap foi cerca de 110 vezes mais rápido que
write() de 1 byte por chamada, porque o write() paga uma chamada de sistema
por byte e o mmap faz apenas uma escrita em memória. Contra write() de uma
página por chamada, a vantagem foi pequena (1,3x a 1,8x), pois ambos acabam
no cache de páginas do kernel. Ao incluir fsync/msync, o tempo é dominado
pela gravação real em disco e a diferença diminui. Os valores dependem da
máquina e do ambiente (este foi um contêiner), então só a ordem de grandeza
deve ser considerada.

#include <sys/stat.h>

#define MAGICO 0x4D454D31
#define VEZ_PROG1 1
#define VEZ_PROG2 2
#define VEZ_FIM 3

struct canal {
    int pronto;
    int vez;
    char texto[256];
};

static void esperaVez(struct canal *c, int quem);

int main (void)
{
    struct canal *c;
    int fd;

    unlink(ARQUIVO);

    fd = open(ARQUIVO, O_RDWR | O_CREAT | O_TRUNC, 0644);
    if (fd == -1) {
        perror("open");
        exit(1);
    }

    if (ftruncate(fd, sizeof(struct canal)) == -1) {
        perror("ftruncate");
        exit(1);
    }

    c = mmap(0, sizeof(struct canal), PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (c == MAP_FAILED) {
        perror("mmap");
        exit(1);
    }

    close(fd);

    __atomic_store_n(&c->vez, VEZ_PROG1, __ATOMIC_RELEASE);
    __atomic_store_n(&c->pronto, MAGICO, __ATOMIC_RELEASE);

    printf("prog1 (pid %d): arquivo %s criado e mapeado em %p\n", getpid(), ARQUIVO, (void *) c);
    fflush(stdout);

    for (int r = 1; r <= RODADAS; r++) {
        esperaVez(c, VEZ_PROG1);
        snprintf(c->texto, sizeof c->texto, "rodada %d: mensagem do prog1 (pid %d)", r, getpid());
        printf("prog1 escreveu: \"%s\"\n", c->texto);
        fflush(stdout);
        __atomic_store_n(&c->vez, VEZ_PROG2, __ATOMIC_RELEASE);

        esperaVez(c, VEZ_PROG1);
        printf("prog1 leu:      \"%s\"\n", c->texto);
        fflush(stdout);
    }

    __atomic_store_n(&c->vez, VEZ_FIM, __ATOMIC_RELEASE);
    printf("prog1: %d rodadas concluidas, encerrando\n", RODADAS);

    munmap(c, sizeof(struct canal));
    return 0;
}

static void esperaVez(struct canal *c, int quem)
{
    while (__atomic_load_n(&c->vez, __ATOMIC_ACQUIRE) != quem)
        usleep(10000);
}

static void mostraMapeamento(const char *rotulo, void *addr);

int main (int argc, char *argv[])
{
    struct stat sb;
    long pagesize = sysconf(_SC_PAGESIZE);
    long mapeado;
    size_t tamMap;
    char *p, *reserva, *alvo;
    const char *modo;
    int fd;

    if (argc < 3) {
        fprintf(stderr, "Uso: %s arquivo dentro|fora|alem|guarda\n", argv[0]);
        exit(1);
    }

    modo = argv[2];

    fd = open(argv[1], O_RDWR);
    if (fd == -1) {
        perror("open");
        exit(1);
    }

    if (fstat(fd, &sb) == -1) {
        perror("fstat");
        exit(1);
    }

    mapeado = ((sb.st_size + pagesize - 1) / pagesize) * pagesize;
    tamMap = sb.st_size;
    if (strcmp(modo, "alem") == 0)
        tamMap = mapeado + pagesize;

    if (strcmp(modo, "guarda") == 0) {

        if (reserva == MAP_FAILED) {
            perror("mmap reserva");
            exit(1);
        }
        p = mmap(reserva, sb.st_size, PROT_READ | PROT_WRITE, MAP_SHARED | MAP_FIXED, fd, 0);
    } else {
        p = mmap(0, tamMap, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    }

    if (p == MAP_FAILED) {
        perror("mmap");
        exit(1);
    }

    printf("arquivo: %ld bytes | pedido ao mmap: %zu bytes | paginas mapeadas do arquivo: %zu\n",
           (long) sb.st_size, tamMap, (tamMap + pagesize - 1) / pagesize);

    mostraMapeamento("mapeamento que contem o byte alvo", alvo);
    printf("escrevendo 'X' em p[%ld] (modo %s)\n", (long) (alvo - p), modo);

    printf("escrita concluida sem erro\n");
    fflush(stdout);

    msync(p, sb.st_size, MS_SYNC);
    munmap(p, tamMap);
    close(fd);

    return 0;
}

static void mostraMapeamento(const char *rotulo, void *addr)
{
    FILE *f = fopen("/proc/self/maps", "r");
    char linha[512];
    unsigned long ini, fim;

    if (f == NULL)
        return;

    while (fgets(linha, sizeof linha, f) != NULL) {
        if (sscanf(linha, "%lx-%lx", &ini, &fim) == 2 && (unsigned long) addr >= ini && (unsigned long) addr < fim) {
            printf("%s: %s", rotulo, linha);
            fclose(f);
            return;
        }
    }

    printf("%s: nenhum mapeamento cobre o endereco %p\n", rotulo, addr);
    fclose(f);
}

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/mman.h>
#include <sys/stat.h>

int main (int argc, char *argv[])
{
    struct stat sb;
    size_t antigo, novo, tamTexto;
    int vezes = 1;
    char *p;
    int fd;

    if (argc < 3) {
        fprintf(stderr, "Uso: %s arquivo texto [vezes]\n", argv[0]);
        exit(1);
    }

    if (argc >= 4)
        vezes = atoi(argv[3]);

    tamTexto = strlen(argv[2]);

    fd = open(argv[1], O_RDWR);
    if (fd == -1) {
        perror("open");
        exit(1);
    }

    if (fstat(fd, &sb) == -1) {
        perror("fstat");
        exit(1);
    }

    if (!S_ISREG(sb.st_mode) || sb.st_size == 0) {
        fprintf(stderr, "%s precisa ser um arquivo regular nao vazio\n", argv[1]);
        exit(1);
    }

    antigo = sb.st_size;
    novo = antigo + tamTexto * vezes;

    p = mmap(0, antigo, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror("mmap");
        exit(1);
    }

    printf("mapeamento inicial: %p com %zu bytes\n", (void *) p, antigo);

    if (ftruncate(fd, novo) == -1) {
        perror("ftruncate");
        exit(1);
    }

    if (munmap(p, antigo) == -1) {
        perror("munmap");
        exit(1);
    }

    p = mmap(0, novo, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror("mmap novo");
        exit(1);
    }

    printf("novo mapeamento:    %p com %zu bytes\n", (void *) p, novo);

    for (int i = 0; i < vezes; i++)
        memcpy(p + antigo + (size_t) i * tamTexto, argv[2], tamTexto);

    if (msync(p, novo, MS_SYNC) == -1) {
        perror("msync");
        exit(1);
    }

    munmap(p, novo);
    close(fd);

    printf("arquivo passou de %zu para %zu bytes\n", antigo, novo);

    return 0;
}

#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/mman.h>
#include <sys/stat.h>

int main (int argc, char *argv[])
{
    struct stat sb;
    size_t antigo, novo, tamTexto;
    int vezes = 1;
    char *p, *q;
    int fd;

    if (argc < 3) {
        fprintf(stderr, "Uso: %s arquivo texto [vezes]\n", argv[0]);
        exit(1);
    }

    if (argc >= 4)
        vezes = atoi(argv[3]);

    tamTexto = strlen(argv[2]);

    fd = open(argv[1], O_RDWR);
    if (fd == -1) {
        perror("open");
        exit(1);
    }

    if (fstat(fd, &sb) == -1) {
        perror("fstat");
        exit(1);
    }

    if (!S_ISREG(sb.st_mode) || sb.st_size == 0) {
        fprintf(stderr, "%s precisa ser um arquivo regular nao vazio\n", argv[1]);
        exit(1);
    }

    antigo = sb.st_size;
    novo = antigo + tamTexto * vezes;

    p = mmap(0, antigo, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);

    if (ftruncate(fd, novo) == -1) {
        perror("ftruncate");
        exit(1);

    q = mremap(p, antigo, novo, MREMAP_MAYMOVE);
    if (q == MAP_FAILED) {
        perror("mremap");
        exit(1);

    printf("apos mremap:        %p com %zu bytes (%s)\n", (void *) q, novo, q == p ? "mesmo endereco" : "o mapeamento mudou de endereco");

    for (int i = 0; i < vezes; i++)
        memcpy(p + antigo + (size_t) i * tamTexto, argv[2], tamTexto);

    if (msync(p, novo, MS_SYNC) == -1) {
        perror("msync");
        exit(1);
    }

    munmap(p, novo);
    close(fd);

    printf("arquivo passou de %zu para %zu bytes\n", antigo, novo);

    return 0;
}
